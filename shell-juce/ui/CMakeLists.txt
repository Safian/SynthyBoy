# synthyboy_ui_model   pure C++17 view-model (no JUCE): key resolution, cap dimming, gestures, assign flow, A/B
# synthyboy_ui        the JUCE Components (kit + editor + every panels/*.cpp) as a host independent static library: the JUCE plug-in, the Kick app
#                      and the tests link it. It needs no AudioProcessor, no plug-in client, no CLAP bridge, no OpenGL; the host supplies a UiHost
#                      (UiHost.h) and with it a ParameterStore (ParameterStore.h). Contract and dependency list: shell-juce/ui/README.md.
#                      Panels are registered EXPLICITLY: PanelRegistration.cpp is generated here from the SYNTHYBOY_REGISTER_PANEL lines of panels/*.cpp
#                      (a new panel is one new file in panels/, picked up by the glob below), no static initialiser, nothing for a linker to drop.

set(UI_DIR "${CMAKE_CURRENT_LIST_DIR}")

add_library(synthyboy_ui_model STATIC
    ${UI_DIR}/model/ControlModel.cpp
    ${UI_DIR}/model/GestureModel.cpp
    ${UI_DIR}/model/FxModel.cpp
    ${UI_DIR}/model/FxKeySet.cpp
    ${UI_DIR}/model/ModModel.cpp)
target_include_directories(synthyboy_ui_model PUBLIC ${UI_DIR}/model)
target_link_libraries(synthyboy_ui_model PUBLIC synthyboy_core)
set_target_properties(synthyboy_ui_model PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(synthyboy_ui_model PRIVATE /W4)
else()
    target_compile_options(synthyboy_ui_model PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

# ---- explicit panel registration ------------------------------------------------------------------------------------------------------------
file(GLOB SYNTHYBOY_UI_PANEL_SOURCES CONFIGURE_DEPENDS ${UI_DIR}/panels/*.cpp)
set(_registerCalls "")
set(_registerDecls "")
foreach(panelSource ${SYNTHYBOY_UI_PANEL_SOURCES})
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${panelSource}") # a registration line added to an existing file
    file(STRINGS "${panelSource}" _lines REGEX "^SYNTHYBOY_REGISTER_PANEL[ ]*\\(")
    foreach(_line ${_lines})
        string(REGEX REPLACE "^SYNTHYBOY_REGISTER_PANEL[ ]*\\([ ]*([A-Za-z0-9_]+)[ ]*,.*$" "\\1" _class "${_line}")
        string(APPEND _registerDecls "void registerPanel_${_class} (PanelRegistry&);\n")
        string(APPEND _registerCalls "    registerPanel_${_class} (registry);\n")
    endforeach()
endforeach()
set(_registrationFile "${CMAKE_CURRENT_BINARY_DIR}/PanelRegistration.cpp")
set(_registrationText "// Generated by shell-juce/ui/CMakeLists.txt from the SYNTHYBOY_REGISTER_PANEL lines of panels/*.cpp. Do not edit.\n#include \"Panel.h\"\n\nnamespace synthyboy::ui\n{\n${_registerDecls}\nvoid registerAllPanels (PanelRegistry& registry)\n{\n${_registerCalls}}\n} // namespace synthyboy::ui\n")
set(_registrationOld "")
if(EXISTS "${_registrationFile}")
    file(READ "${_registrationFile}" _registrationOld)
endif()
if(NOT _registrationOld STREQUAL _registrationText)
    file(WRITE "${_registrationFile}" "${_registrationText}")
endif()

add_library(synthyboy_ui STATIC
    ${UI_DIR}/Theme.cpp
    ${UI_DIR}/UiContext.cpp
    ${UI_DIR}/ParameterStore.cpp
    ${UI_DIR}/ParamHandle.cpp
    ${UI_DIR}/Overlays.cpp
    ${UI_DIR}/ContentIO.cpp
    ${UI_DIR}/LiveBus.cpp
    ${UI_DIR}/LibraryMenu.cpp
    ${UI_DIR}/ContentSheets.cpp
    ${UI_DIR}/ParamControls.cpp
    ${UI_DIR}/Sheet.cpp
    ${UI_DIR}/ModuleSheet.cpp
    ${UI_DIR}/FxRackOps.cpp
    ${UI_DIR}/GestureKit.cpp
    ${UI_DIR}/ModSourceBar.cpp
    ${UI_DIR}/Panel.cpp
    ${UI_DIR}/PanelHost.cpp
    ${UI_DIR}/Strip.cpp
    ${UI_DIR}/Editor.cpp
    ${SYNTHYBOY_UI_PANEL_SOURCES}
    "${_registrationFile}")
# The includes of the sources are relative to ui/ ("panels/Foo.h", "model/ModModel.h"), the includes of consumers to shell-juce/ ("ui/Editor.h")
target_include_directories(synthyboy_ui PUBLIC ${UI_DIR} ${UI_DIR}/..)
target_compile_features(synthyboy_ui PUBLIC cxx_std_17)
set_target_properties(synthyboy_ui PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)

# Dependency list (README.md "Dependencies"): the core library (it carries the model, content, import, shapes, wavetable and sample packages the Components
# use) and the JUCE modules juce_gui_basics (with juce_graphics, juce_events, juce_core below it), juce_audio_basics, juce_audio_formats (decoding of user
# samples) and juce_audio_utils (MidiKeyboardComponent). JUCE 7.0.12 and 8.x both work (shell-juce/ui/JuceCompat.h).
# juce_audio_processors is NOT used by the library, although juce_audio_utils pulls it in.
target_link_libraries(synthyboy_ui
    PUBLIC  synthyboy_ui_model synthyboy_core
            juce::juce_gui_basics juce::juce_audio_basics juce::juce_audio_formats juce::juce_audio_utils
            Threads::Threads)
# the JUCE module configuration the sources are compiled with (the headers of juce_audio_utils include juce_gui_extra, whose web browser needs nothing here)
target_compile_definitions(synthyboy_ui PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
if(TARGET juce::juce_recommended_config_flags)
    target_link_libraries(synthyboy_ui PRIVATE juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
endif()

    synthyboy_add_test(synthyboy_test_ui_model tests/test_ui_model.cpp LINK synthyboy_ui_model)
